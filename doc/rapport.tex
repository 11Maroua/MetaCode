\section{Introduction}

Le Set Packing Problem (SPP) est un problème d'optimisation combinatoire NP-difficile.

\subsection{Définition}

Maximiser :
\begin{equation}
    \max_{x \in \{0,1\}^n} \sum_{j=1}^{n} c_j x_j
\end{equation}

Sous contraintes :
\begin{equation}
    \sum_{j=1}^{n} a_{ij} x_j \leq 1, \quad \forall i \in \{1,\ldots,m\}
\end{equation}

\section{EI2 : GRASP et Reactive GRASP}

\subsection{GRASP Standard}

Phase construction avec RCL :
\begin{equation}
    \text{RCL} = \{j : u_j \geq u_{\min} + \alpha (u_{\max} - u_{\min})\}
\end{equation}

\subsection{Reactive GRASP}

Adaptation automatique de $\alpha$ par qualité :
\begin{equation}
    p_k = \frac{q_k}{\sum_{i} q_i}, \quad q_k = \frac{\bar{z}_k - z_{\text{worst}}}{z_{\text{best}} - z_{\text{worst}}}
\end{equation}
